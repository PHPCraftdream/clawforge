_clawforge_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
    if [[ $skip -eq 1 ]]; then skip=0; continue; fi
    if [[ "$w" == "--app" ]]; then skip=1; continue; fi
    cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
    if [[ "$prev" == "--app" ]]; then
    COMPREPLY=( $(compgen -W "$("${COMP_WORDS[0]}" list --json --no-status 2>/dev/null | grep -o '"name":"[^"]*"' | cut -d'"' -f4 | grep -v '^[.]')" -- "$cur") )
    return
  fi
    if reply="$(_clawforge_lookup "top")"; then COMPREPLY=( $(compgen -W "$reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The word after the command scopes an option's values; the option itself, last of
  # `between`, is not a scope.
  if (( ${#between[@]} > 0 )) && [[ "${between[0]}" != "$prev" ]]; then scope="${between[0]}"; fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if reply="$(_clawforge_lookup "first $cmd")"; then COMPREPLY=( $(compgen -W "$reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # An option's own values, keyed command + scope + option, take the value position.
  if [[ "$prev" == --* ]] && reply="$(_clawforge_lookup "values ${cmd}${scope}${prev}")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
    return
  fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if reply="$(_clawforge_lookup "after $cmd ${between[0]}")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
  elif reply="$(_clawforge_lookup "after $cmd *")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}
complete -F _clawforge_complete clawforge
complete -F _clawforge_complete ./clawforge

_clawforge_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
    if [[ $skip -eq 1 ]]; then skip=0; continue; fi
    if [[ "$w" == "--app" ]]; then skip=1; continue; fi
    cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
    if [[ "$prev" == "--app" ]]; then
    COMPREPLY=( $(compgen -W "$("${COMP_WORDS[0]}" list --json --no-status 2>/dev/null | grep -o '"name":"[^"]*"' | cut -d'"' -f4 | grep -v '^[.]')" -- "$cur") )
    return
  fi
    if reply="$(_clawforge_lookup "top")"; then COMPREPLY=( $(compgen -W "$reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The word after the command scopes an option's values; the option itself, last of
  # `between`, is not a scope.
  if (( ${#between[@]} > 0 )) && [[ "${between[0]}" != "$prev" ]]; then scope="${between[0]}"; fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if reply="$(_clawforge_lookup "first $cmd")"; then COMPREPLY=( $(compgen -W "$reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # An option's own values, keyed command + scope + option, take the value position.
  if [[ "$prev" == --* ]] && reply="$(_clawforge_lookup "values ${cmd}${scope}${prev}")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
    return
  fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if reply="$(_clawforge_lookup "after $cmd ${between[0]}")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
  elif reply="$(_clawforge_lookup "after $cmd *")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}
complete -F _clawforge_complete clawforge
complete -F _clawforge_complete ./clawforge

$clawforgeCompleter = {
  param($wordToComplete, $commandAst, $cursorPosition)
  $tokens = @($commandAst.CommandElements | ForEach-Object { $_.Extent.Text })
  $rest = @(if ($tokens.Count -gt 1) { @($tokens[1..($tokens.Count - 1)]) } else { @() })
  # Only the words BEFORE the word being completed pick the command — the partial word
  # itself is not a typed command (R32-02: 'clawforge sta<Tab>' saw the command "sta").
  $scan = $rest
  if ($wordToComplete -ne '') {
    $scan = @(if ($rest.Count -gt 1) { @($rest[0..($rest.Count - 2)]) } else { @() })
  }
  $prev = $null
  if ($scan.Count -ge 1) { $prev = $scan[$scan.Count - 1] }
  # --app is positional and takes a value: the pair is skipped, so it is never read as the
  # command (and a value that looks like a command name does not shadow the real one).
  $i = 0
  while ($clawforgeApp -and $i -lt $scan.Count -and $scan[$i] -eq '--app') { $i = $i + 2 }
  $candidates = @()
  if ($i -ge $scan.Count) {
    if ($clawforgeApp -and $prev -eq '--app') {
      # --app's own value, lazily: only while no command word has been typed — --app must
      # come before the command, so past one the command's flags return (R33-10). Whichever
      # of the system-wide command or the checkout shim was typed makes the call.
      $names = @(try { & $tokens[0] list --json --no-status 2>$null | ConvertFrom-Json | ForEach-Object { $_.name } | Where-Object { $_ -notlike '.*' } } catch { @() })
      $names | Where-Object { $_ -like "$wordToComplete*" } | ForEach-Object {
        [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
      }
      return
    }
    $candidates = @($clawforgeTop)
  } else {
    $cmd = $scan[$i]
    $between = @()
    if ($scan.Count - $i - 1 -gt 0) { $between = @($scan[($i + 1)..($scan.Count - 1)]) }
    # An option's values are keyed command + scope + option, where the scope is the word
    # after the command; the option itself (last of `between` after a trailing space) is
    # not a scope.
    $scope = ''
    if ($between.Count -gt 0 -and $between[0] -ne $prev) { $scope = $between[0] }
    $key = "$cmd$scope$prev"
    if ($prev -ne $null -and $prev.StartsWith('--') -and $clawforgeValues.ContainsKey($key)) {
      $candidates = @($clawforgeValues[$key])
    } elseif ($between.Count -eq 0) {
      # An unknown command reaches `first` with nothing under that key: the @() keeps the
      # type an array and the filter below drops the $null — nothing is offered.
      $candidates = @($clawforgeFirst[$cmd])
    } else {
      $afterKey = "$cmd $($between[0])"
      if ($clawforgeAfter.ContainsKey($afterKey)) {
        $candidates = @($clawforgeAfter[$afterKey])
      } else {
        $candidates = @($clawforgeAfter["$cmd *"])
      }
    }
  }
  $candidates | Where-Object { $_ -like "$wordToComplete*" } | Sort-Object -Unique | ForEach-Object {
    [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
  }
}

_clawforge_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
        cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
        if reply="$(_clawforge_lookup "top")"; then COMPREPLY=( $(compgen -W "$reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The word after the command scopes an option's values; the option itself, last of
  # `between`, is not a scope.
  if (( ${#between[@]} > 0 )) && [[ "${between[0]}" != "$prev" ]]; then scope="${between[0]}"; fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if reply="$(_clawforge_lookup "first $cmd")"; then COMPREPLY=( $(compgen -W "$reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # An option's own values, keyed command + scope + option, take the value position.
  if [[ "$prev" == --* ]] && reply="$(_clawforge_lookup "values ${cmd}${scope}${prev}")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
    return
  fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if reply="$(_clawforge_lookup "after $cmd ${between[0]}")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
  elif reply="$(_clawforge_lookup "after $cmd *")"; then
    COMPREPLY=( $(compgen -W "$reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}

$clawforgeCompleter = {
  param($wordToComplete, $commandAst, $cursorPosition)
  $tokens = @($commandAst.CommandElements | ForEach-Object { $_.Extent.Text })
  $rest = @(if ($tokens.Count -gt 1) { @($tokens[1..($tokens.Count - 1)]) } else { @() })
  # Only the words BEFORE the word being completed pick the command — the partial word
  # itself is not a typed command (R32-02: 'clawforge sta<Tab>' saw the command "sta").
  $scan = $rest
  if ($wordToComplete -ne '') {
    $scan = @(if ($rest.Count -gt 1) { @($rest[0..($rest.Count - 2)]) } else { @() })
  }
  $prev = $null
  if ($scan.Count -ge 1) { $prev = $scan[$scan.Count - 1] }
  # --app is positional and takes a value: the pair is skipped, so it is never read as the
  # command (and a value that looks like a command name does not shadow the real one).
  $i = 0
  while ($clawforgeApp -and $i -lt $scan.Count -and $scan[$i] -eq '--app') { $i = $i + 2 }
  $candidates = @()
  if ($i -ge $scan.Count) {
    if ($clawforgeApp -and $prev -eq '--app') {
      # --app's own value, lazily: only while no command word has been typed — --app must
      # come before the command, so past one the command's flags return (R33-10). Whichever
      # of the system-wide command or the checkout shim was typed makes the call.
      $names = @(try { & $tokens[0] list --json --no-status 2>$null | ConvertFrom-Json | ForEach-Object { $_.name } | Where-Object { $_ -notlike '.*' } } catch { @() })
      $names | Where-Object { $_ -like "$wordToComplete*" } | ForEach-Object {
        [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
      }
      return
    }
    $candidates = @($clawforgeTop)
  } else {
    $cmd = $scan[$i]
    $between = @()
    if ($scan.Count - $i - 1 -gt 0) { $between = @($scan[($i + 1)..($scan.Count - 1)]) }
    # An option's values are keyed command + scope + option, where the scope is the word
    # after the command; the option itself (last of `between` after a trailing space) is
    # not a scope.
    $scope = ''
    if ($between.Count -gt 0 -and $between[0] -ne $prev) { $scope = $between[0] }
    $key = "$cmd$scope$prev"
    if ($prev -ne $null -and $prev.StartsWith('--') -and $clawforgeValues.ContainsKey($key)) {
      $candidates = @($clawforgeValues[$key])
    } elseif ($between.Count -eq 0) {
      # An unknown command reaches `first` with nothing under that key: the @() keeps the
      # type an array and the filter below drops the $null — nothing is offered.
      $candidates = @($clawforgeFirst[$cmd])
    } else {
      $afterKey = "$cmd $($between[0])"
      if ($clawforgeAfter.ContainsKey($afterKey)) {
        $candidates = @($clawforgeAfter[$afterKey])
      } else {
        $candidates = @($clawforgeAfter["$cmd *"])
      }
    }
  }
  $candidates | Where-Object { $_ -like "$wordToComplete*" } | Sort-Object -Unique | ForEach-Object {
    [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
  }
}
Register-ArgumentCompleter -Native -CommandName clawforge, ./clawforge -ScriptBlock $clawforgeCompleter
